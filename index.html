/************************************************* * 철도비서 - Google Apps Script 서버 (KORAIL.gs) * * 기능 * 1. 회원가입 / 로그인 (사원번호 중복 확인, 동시 가입 방지) * 2. DIA_DB 시트의 DIA 목록 불러오기 (순환 순서) * 3. 사용자별 근무표 설정(선택 DIA + 기준일) 저장/불러오기 * 4. JSON / JSONP 응답, 한국시간(KST) 저장 * * 시트 구성 (없으면 자동 생성) * 회원정보 : 사원번호 | 이름 | 비밀번호 | 가입일시 * DIA_DB : 순서 | DIA | 구분 | 시작 | 종료 | 열차1 | 열차2 * 사용자설정 : 사원번호 | DIA | 기준일 | 수정일시 *************************************************/ const SHEET_NAME = '회원정보'; const DIA_SHEET_NAME = 'DIA_DB'; const SETTINGS_SHEET_NAME = '사용자설정'; /* ================================================ GET 요청 처리 ================================================ */ function doGet(e) { e = e || {}; const params = e.parameter || {}; const callback = String(params.callback || '').trim(); let result; try { const action = String(params.action || '').trim(); if (action === 'test') { result = { success: true, message: '철도비서 서버 연결 성공', time: getKST() }; } else if (action === 'signup') { result = signup(params); } else if (action === 'login') { result = login(params); } else if (action === 'getDia') { result = getDiaList(); } else if (action === 'getSettings') { result = getUserSettings(params); } else if (action === 'saveSettings') { result = saveUserSettings(params); } else { result = { success: false, message: '잘못된 요청입니다.', errorCode: 'INVALID_ACTION' }; } } catch (error) { result = { success: false, message: '서버 오류가 발생했습니다.', errorCode: 'SERVER_ERROR', error: String(error && error.message ? error.message : error) }; } return createResponse(result, callback); } /* ================================================ 회원가입 ================================================ */ function signup(params) { try { params = params || {}; const employeeId = String(params.employeeId || '').trim(); const name = String(params.name || '').trim(); const password = String(params.password || ''); if (!employeeId) { return { success: false, message: '사원번호를 입력해주세요.', errorCode: 'EMPTY_EMPLOYEE_ID' }; } if (!name) { return { success: false, message: '이름을 입력해주세요.', errorCode: 'EMPTY_NAME' }; } if (!password) { return { success: false, message: '비밀번호를 입력해주세요.', errorCode: 'EMPTY_PASSWORD' }; } if (employeeId.length > 30) { return { success: false, message: '사원번호가 너무 깁니다.', errorCode: 'INVALID_EMPLOYEE_ID' }; } if (name.length > 50) { return { success: false, message: '이름이 너무 깁니다.', errorCode: 'INVALID_NAME' }; } if (password.length > 100) { return { success: false, message: '비밀번호가 너무 깁니다.', errorCode: 'INVALID_PASSWORD' }; } const sheet = getMemberSheet(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const existingIds = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues(); for (let i = 0; i < existingIds.length; i++) { const existingId = String(existingIds[i][0] || '').trim(); if (existingId && existingId === employeeId) { return { success: false, message: '이미 가입된 사원번호입니다.', errorCode: 'DUPLICATE_EMPLOYEE_ID' }; } } } const now = getKST(); sheet.appendRow([employeeId, name, password, now]); const savedRow = sheet.getLastRow(); const savedEmployeeId = String(sheet.getRange(savedRow, 1).getDisplayValue()).trim(); if (savedEmployeeId !== employeeId) { return { success: false, message: '회원정보 저장을 확인하지 못했습니다.', errorCode: 'SAVE_VERIFY_FAILED' }; } return { success: true, message: '회원가입이 완료되었습니다.', employeeId: employeeId, name: name, signupDate: now }; } finally { lock.releaseLock(); } } catch (error) { return { success: false, message: '회원가입 처리 중 오류가 발생했습니다.', errorCode: 'SIGNUP_ERROR', error: String(error && error.message ? error.message : error) }; } } /* ================================================ 로그인 ================================================ */ function login(params) { try { params = params || {}; const employeeId = String(params.employeeId || '').trim(); const password = String(params.password || ''); if (!employeeId) { return { success: false, message: '사원번호를 입력해주세요.', errorCode: 'EMPTY_EMPLOYEE_ID' }; } if (!password) { return { success: false, message: '비밀번호를 입력해주세요.', errorCode: 'EMPTY_PASSWORD' }; } const sheet = getMemberSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return { success: false, message: '가입된 회원정보가 없습니다.', errorCode: 'NO_MEMBERS' }; } const rows = sheet.getRange(2, 1, lastRow - 1, 4).getDisplayValues(); for (let i = 0; i < rows.length; i++) { const storedEmployeeId = String(rows[i][0] || '').trim(); const storedName = String(rows[i][1] || '').trim(); const storedPassword = String(rows[i][2] || ''); if (storedEmployeeId === employeeId && storedPassword === password) { return { success: true, message: '로그인 성공', employeeId: storedEmployeeId, name: storedName, token: makeToken(storedEmployeeId, storedPassword) }; } } return { success: false, message: '사원번호 또는 비밀번호가 올바르지 않습니다.', errorCode: 'LOGIN_FAILED' }; } catch (error) { return { success: false, message: '로그인 처리 중 오류가 발생했습니다.', errorCode: 'LOGIN_ERROR', error: String(error && error.message ? error.message : error) }; } } /* ================================================ 인증 토큰 - 비밀번호를 기기에 저장하지 않고도 설정을 저장/조회하기 위한 값 - 사원번호 + 비밀번호로 만든 서명이라 비밀번호가 바뀌면 무효가 됨 ================================================ */ function getSecret() { const props = PropertiesService.getScriptProperties(); let secret = props.getProperty('TOKEN_SECRET'); if (!secret) { secret = Utilities.getUuid() + Utilities.getUuid(); props.setProperty('TOKEN_SECRET', secret); } return secret; } function makeToken(employeeId, password) { const sig = Utilities.computeHmacSha256Signature( employeeId + '|' + password, getSecret() ); return Utilities.base64EncodeWebSafe(sig); } // 토큰이 맞으면 true function checkToken(employeeId, token) { employeeId = String(employeeId || '').trim(); token = String(token || ''); if (!employeeId || !token) return false; const sheet = getMemberSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) return false; const rows = sheet.getRange(2, 1, lastRow - 1, 3).getDisplayValues(); for (let i = 0; i < rows.length; i++) { if (String(rows[i][0] || '').trim() === employeeId) { return makeToken(employeeId, String(rows[i][2] || '')) === token; } } return false; } function authFailed() { return { success: false, message: '로그인 정보가 올바르지 않습니다. 다시 로그인해주세요.', errorCode: 'AUTH_FAILED' }; } /* ================================================ DIA 목록 (DIA_DB 시트) ================================================ */ // 1행 제목으로 열 위치를 찾음 (못 찾으면 fallback 열 사용, -1이면 없음) function findCol(header, names, fallback) { for (let i = 0; i < header.length; i++) { const h = String(header[i] || '').replace(/\s+/g, '').toUpperCase(); if (names.indexOf(h) !== -1) return i; } return fallback; } function getDiaList() { try { const sheet = getDiaSheet(); const lastRow = sheet.getLastRow(); const lastCol = Math.max(sheet.getLastColumn(), 1); if (lastRow < 2) { return { success: true, list: [], count: 0 }; } const values = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues(); const header = values[0]; const cOrder = findCol(header, ['순서', '근무순서', '번호', 'NO'], -1); const cDia = findCol(header, ['DIA', '다이아', 'DIA번호', '다이아번호'], 2); const cType = findCol(header, ['구분', '근무구분', '근무'], -1); const cStart = findCol(header, ['시작', '시작시간', '출근'], -1); const cEnd = findCol(header, ['종료', '종료시간', '퇴근'], -1); const cTrain1 = findCol(header, ['열차1', '열차번호1'], -1); const cTrain2 = findCol(header, ['열차2', '열차번호2'], -1); const cell = function (row, c) { return c < 0 ? '' : String(row[c] || '').trim(); }; const list = []; for (let i = 1; i < values.length; i++) { const row = values[i]; const dia = cell(row, cDia); const type = cell(row, cType); // DIA 번호도 구분도 없는 빈 줄은 건너뜀 (비번/휴무처럼 DIA 번호가 비어 있는 줄은 포함) if (!dia && !type) continue; const orderNum = parseFloat(cell(row, cOrder)); const order = isNaN(orderNum) ? i : orderNum; list.push({ order: order, seq: String(order), dia: dia, type: type, start: normTime(cell(row, cStart)), end: normTime(cell(row, cEnd)), train1: cell(row, cTrain1), train2: cell(row, cTrain2) }); } list.sort(function (a, b) { return a.order - b.order; }); const result = { success: true, list: list, count: list.length }; if (cType < 0) { result.warning = "DIA_DB 시트 1행에 '구분' 제목이 없어 달력에 구분을 표시할 수 없습니다."; } return result; } catch (error) { return { success: false, message: 'DIA 목록을 불러오지 못했습니다.', errorCode: 'DIA_ERROR', error: String(error && error.message ? error.message : error) }; } } // 시트가 시간 형식으로 바꿔 놓은 값도 HH:mm 으로 통일 function normTime(value) { const s = String(value || '').trim(); const m = s.match(/(\d{1,2}):(\d{2})(?::\d{2})?\s*(AM|PM|오전|오후)?/i); if (!m) return s; let h = parseInt(m[1], 10); const ap = m[3] ? m[3].toUpperCase() : ''; if ((ap === 'PM' || ap === '오후') && h < 12) h += 12; if ((ap === 'AM' || ap === '오전') && h === 12) h = 0; return ('0' + h).slice(-2) + ':' + m[2]; } // 시트의 열 개수가 부족하면 늘려서 범위 오류를 막음 function ensureColumns(sheet, count) { const max = sheet.getMaxColumns(); if (max < count) { sheet.insertColumnsAfter(max, count - max); } } function getDiaSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error('Google Sheets와 연결된 Apps Script가 아닙니다.'); } let sheet = spreadsheet.getSheetByName(DIA_SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(DIA_SHEET_NAME); } ensureColumns(sheet, 7); if (sheet.getLastRow() === 0) { // 시간/번호가 자동 변환되지 않도록 문자열 형식으로 고정 sheet.getRange('A:G').setNumberFormat('@'); sheet.getRange(1, 1, 1, 7).setValues([ ['순서', 'DIA', '구분', '시작', '종료', '열차1', '열차2'] ]); // 처음 한 번만 들어가는 기본 DIA (시트에서 자유롭게 수정/추가하세요) sheet.getRange(2, 1, 10, 7).setValues([ ['1', '12602', '주간', '07:43', '18:46', '6127', '6128'], ['2', '12404', '야간', '18:20', '07:49', '717', '3252'], ['3', '12601', '주간', '07:30', '18:38', 'L6121', '6122'], ['4', '12402', '주간', '12:59', '20:59', '3226', '3227'], ['5', '12581', '야간', '22:06', '10:24', '9999', 'L6129'], ['6', '12101', '주간', '08:32', '19:06', '1279', '1208'], ['7', '12551', '주간', '07:45', '17:47', 'K/M6129', ''], ['8', '12201', '주간', '09:10', '18:10', '3224', '3223'], ['9', '12401', '주간', '10:45', '20:31', '3334', 'D1208'], ['10', '12403', '야간', '18:20', '08:29', '717', '3262'] ]); } return sheet; } /* ================================================ 사용자 설정 (선택 DIA + 기준일) ================================================ */ function getSettingsSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error('Google Sheets와 연결된 Apps Script가 아닙니다.'); } let sheet = spreadsheet.getSheetByName(SETTINGS_SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(SETTINGS_SHEET_NAME); } ensureColumns(sheet, 4); const headerValues = [['사원번호', '선택순서', '기준일', '수정일시']]; if (sheet.getLastRow() === 0) { sheet.getRange('A:D').setNumberFormat('@'); sheet.getRange(1, 1, 1, 4).setValues(headerValues); } else { const header = sheet.getRange(1, 1, 1, 4).getDisplayValues()[0]; if (header[1] !== '선택순서') { sheet.getRange(1, 1, 1, 4).setValues(headerValues); } } return sheet; } function getUserSettings(params) { try { params = params || {}; const employeeId = String(params.employeeId || '').trim(); if (!checkToken(employeeId, params.token)) return authFailed(); const sheet = getSettingsSheet(); const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const rows = sheet.getRange(2, 1, lastRow - 1, 3).getDisplayValues(); for (let i = 0; i < rows.length; i++) { if (String(rows[i][0] || '').trim() === employeeId) { const saved = String(rows[i][1] || '').trim(); let seq = ''; const diaResult = getDiaList(); if (diaResult.success && saved) { // 저장값이 순서이면 그대로, 예전에 저장한 DIA 번호이면 순서로 변환 const found = diaResult.list.filter(function (it) { return it.seq === saved; })[0] || diaResult.list.filter(function (it) { return it.dia === saved; })[0]; if (found) seq = found.seq; } return { success: true, seq: seq, anchor: String(rows[i][2] || '').trim() }; } } } // 저장된 설정이 없는 경우 return { success: true, seq: '', anchor: '' }; } catch (error) { return { success: false, message: '설정을 불러오지 못했습니다.', errorCode: 'GET_SETTINGS_ERROR', error: String(error && error.message ? error.message : error) }; } } function saveUserSettings(params) { try { params = params || {}; const employeeId = String(params.employeeId || '').trim(); if (!checkToken(employeeId, params.token)) return authFailed(); const seq = String(params.seq || '').trim(); const anchor = String(params.anchor || '').trim(); if (!seq) { return { success: false, message: 'DIA를 선택해주세요.', errorCode: 'EMPTY_SEQ' }; } if (!/^\d{4}-\d{2}-\d{2}$/.test(anchor)) { return { success: false, message: '기준일 형식이 올바르지 않습니다.', errorCode: 'INVALID_ANCHOR' }; } // DIA_DB에 없는 순서는 저장하지 않음 const diaResult = getDiaList(); if (!diaResult.success) return diaResult; const exists = diaResult.list.some(function (item) { return item.seq === seq; }); if (!exists) { return { success: false, message: 'DIA_DB에 없는 순서입니다.', errorCode: 'UNKNOWN_SEQ' }; } const sheet = getSettingsSheet(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const now = getKST(); const lastRow = sheet.getLastRow(); let targetRow = 0; if (lastRow >= 2) { const ids = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0] || '').trim() === employeeId) { targetRow = i + 2; break; } } } if (!targetRow) { targetRow = lastRow + 1; } sheet.getRange(targetRow, 1, 1, 4).setNumberFormat('@'); sheet.getRange(targetRow, 1, 1, 4).setValues([[employeeId, seq, anchor, now]]); return { success: true, message: '저장되었습니다.', seq: seq, anchor: anchor }; } finally { lock.releaseLock(); } } catch (error) { return { success: false, message: '설정 저장 중 오류가 발생했습니다.', errorCode: 'SAVE_SETTINGS_ERROR', error: String(error && error.message ? error.message : error) }; } } /* ================================================ 회원정보 시트 가져오기 ================================================ */ function getMemberSheet() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error('Google Sheets와 연결된 Apps Script가 아닙니다.'); } let sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { sheet = spreadsheet.insertSheet(SHEET_NAME); } ensureColumns(sheet, 4); const headerValues = [['사원번호', '이름', '비밀번호', '가입일시']]; if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, 4).setValues(headerValues); } const header = sheet.getRange(1, 1, 1, 4).getDisplayValues()[0]; if ( header[0] !== '사원번호' || header[1] !== '이름' || header[2] !== '비밀번호' || header[3] !== '가입일시' ) { sheet.getRange(1, 1, 1, 4).setValues(headerValues); } try { sheet.setColumnWidth(1, 120); sheet.setColumnWidth(2, 100); sheet.setColumnWidth(3, 150); sheet.setColumnWidth(4, 180); } catch (e) { // 열 너비 설정 실패는 무시 } return sheet; } /* ================================================ JSON / JSONP 응답 생성 ================================================ */ function createResponse(result, callback) { const json = JSON.stringify(result); if (callback) { // callback에는 함수명만 허용 const safeCallback = callback.replace(/[^a-zA-Z0-9_$\.]/g, ''); if (safeCallback) { return ContentService .createTextOutput(safeCallback + '(' + json + ');') .setMimeType(ContentService.MimeType.JAVASCRIPT); } } return ContentService .createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } /* ================================================ 한국시간 반환 ================================================ */ function getKST() { return Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss'); } /* ================================================ 테스트 ================================================ */ function testServer() { const result = { success: true, message: '철도비서 서버 연결 성공', time: getKST() }; Logger.log(JSON.stringify(result)); return result; } function testMemberSheet() { const sheet = getMemberSheet(); const result = { success: true, sheetName: sheet.getName(), spreadsheetName: sheet.getParent().getName(), lastRow: sheet.getLastRow(), lastColumn: sheet.getLastColumn(), time: getKST() }; Logger.log(JSON.stringify(result)); return result; } function testDiaSheet() { const result = getDiaList(); Logger.log(JSON.stringify(result)); return result; } /* ================================================ 시트 한 번에 만들기 - Apps Script 편집기에서 이 함수를 선택하고 [실행]을 한 번 누르면 회원정보 / DIA_DB / 사용자설정 시트가 모두 생성됩니다. ================================================ */ function setupSheets() { const result = { success: true, member: getMemberSheet().getName(), dia: getDiaSheet().getName(), settings: getSettingsSheet().getName(), spreadsheetName: SpreadsheetApp.getActiveSpreadsheet().getName(), time: getKST() }; Logger.log(JSON.stringify(result)); return result; }
